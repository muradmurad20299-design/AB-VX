import pandas as pd

def audit_financial_data(file_path):
    # 1. قراءة البيانات من ملف الإكسل أو CSV
    df = pd.read_csv(file_path) # أو read_excel
    df['date'] = pd.to_datetime(df['date']) # تحويل عمود التاريخ لصيغة قابلة للتحليل
    
    anomalies = {} # قاموس لتخزين الأخطاء المكتشفة

    # ---------------------------------------------------------
    # القاعدة الأولى: اكتشاف المدفوعات المكررة
    # البحث عن حركات لنفس المورد، بنفس المبلغ، وتواريخ متقاربة
    # ---------------------------------------------------------
    duplicates = df[df.duplicated(subset=['vendor_id', 'amount'], keep=False)]
    anomalies['duplicate_payments'] = duplicates
    
    # ---------------------------------------------------------
    # القاعدة الثانية: اكتشاف تجزئة المشتريات (Splitting)
    # لنفترض أن سقف الشراء المباشر هو 10,000. الموظف قد يصدر 3 فواتير بقيمة 4000 لنفس المورد خلال أسبوع
    # ---------------------------------------------------------
    # تجميع المبالغ لكل مورد خلال نفس الأسبوع أو الشهر
    weekly_spend = df.groupby(['vendor_id', pd.Grouper(key='date', freq='W-MON')])['amount'].sum().reset_index()
    split_purchases = weekly_spend[weekly_spend['amount'] > 10000] # تحديد الموردين الذين تجاوزوا السقف
    anomalies['split_purchases'] = split_purchases

    # ---------------------------------------------------------
    # القاعدة الثالثة: اكتشاف القيود غير الاعتيادية (أيام العطل)
    # ---------------------------------------------------------
    # استخراج يوم الأسبوع (5=الجمعة، 6=السبت)
    df['day_of_week'] = df['date'].dt.dayofweek
    weekend_transactions = df[df['day_of_week'].isin([5, 6])]
    anomalies['weekend_transactions'] = weekend_transactions
    
    return anomalies

# هكذا يقوم البرنامج بإرجاع تقرير كامل بالمخالفات
